### 3.2 连接流控状态

```c
// 连接级流控状态

struct connection_flow_control {
    // 最大数据量（对端允许）
    u64 max_data;

    // 已确认接收的数据
    u64 data_read;

    // 当前窗口
    u64 window() {
        return max_data - data_read;
    }
};

// 检查是否可以发送
bool can_send(struct connection_flow_control *cfc, u64 bytes) {
    return cfc->window() >= bytes;
}

// 更新 MAX_DATA
void update_max_data(struct connection_flow_control *cfc, u64 new_max) {
    cfc->max_data = new_max;
}
```

### 4.2 流状态

```c
// 流级流控状态

enum stream_state {
    STREAM_IDLE,        // 流已创建，未使用
    STREAM_OPEN,        // 发送 OPEN 帧
    STREAM_SEND,        // 可发送数据
    STREAM_RECV,        // 接收数据
    STREAM_DATA_SENT,   // 数据已发送
    STREAM_RESET,       // 流重置
    STREAM_REMOTE_RESET,// 远程重置
    STREAM_CLOSED       // 流关闭
};

// 单向流 vs 双向流
enum stream_direction {
    STREAM_UNIDIRECTIONAL,  // 单向
    STREAM_BIDIRECTIONAL    // 双向
};

struct stream_flow_control {
    u64 stream_id;

    // 发送相关
    u64 sent_offset;      // 已发送的字节
    u64 max_stream_data;  // 对端允许的最大

    // 接收相关
    u64 recv_offset;      // 已接收的字节
    u64 max_recv_data;    // 本端允许的最大

    bool blocked;         // 是否被流控阻止
};
```

### 4.3 流控检查

```c
// 流级发送检查

bool stream_can_send(struct stream_flow_control *sfc) {
    // 检查方向
    if (!is_sendable_stream(sfc->stream_id))
        return false;

    // 检查 FIN 是否已发送
    if (sfc->fin_sent)
        return false;

    // 检查流控窗口
    if (sfc->sent_offset >= sfc->max_stream_data)
        return false;

    return true;
}

// 更新 MAX_STREAM_DATA
void update_max_stream_data(struct stream_flow_control *sfc,
                            u64 new_limit) {
    u64 old_limit = sfc->max_stream_data;
    sfc->max_stream_data = new_limit;

    // 如果窗口打开，发送数据
    if (new_limit > old_limit && !sfc->blocked) {
        sfc->blocked = false;
        notify_app_to_send_stream(sfc->stream_id);
    }
}
```

### 6.2 Cubic 算法

```
CUBIC 算法（QUIC 默认）：

┌──────────────────────────────────────────────────────────────────────┐
│                        CUBIC 原理                                   │
├──────────────────────────────────────────────────────────────────────┤
│                                                                      │
│   核心思想：                                                        │
│   · 使用三次多项式控制窗口增长                                      │
│   · 在瓶颈处更稳定                                                 │
│   · 快速恢复带宽                                                   │
│                                                                      │
│   W(t) = C * (t - K)^3 + Wmax                                     │
│                                                                      │
│   其中：                                                            │
│   · C = 0.4 (缩放因子)                                             │
│   · t = 当前时间自从上次丢包开始                                    │
│   · K = 窗口从 Wmax 降到 Wmax*β 的时间                             │
│   · Wmax = 上次拥塞前的窗口大小                                    │
│                                                                      │
└──────────────────────────────────────────────────────────────────────┘

CUBIC 状态机：

┌──────────────────────────────────────────────────────────────────────┐
│                        CUBIC 状态                                   │
├──────────────────────────────────────────────────────────────────────┤
│                                                                      │
│   ┌─────────────────────────────────────────────────────────────┐   │
│   │                      慢启动 (Slow Start)                      │   │
│   │                                                             │   │
│   │  CWND 增长：                                                │   │
│   │  CWND += MSS * acked (每个 ACK)                            │   │
│   │                                                             │   │
│   │  结束条件：ssthresh                                         │   │
│   │  CWND >= ssthresh → 转入拥塞避免                            │   │
│   └─────────────────────────────────────────────────────────────┘   │
│                              │                                       │
│                              ▼                                       │
│   ┌─────────────────────────────────────────────────────────────┐   │
│   │                    拥塞避免 (Congestion Avoidance)            │   │
│   │                                                             │   │
│   │  CWND 增长：                                                │   │
│   │  W(t) = C * (t - K)^3 + Wmax                              │   │
│   │                                                             │   │
│   │  接近 Wmax 时增长慢                                         │   │
│   │  远离 Wmax 时增长快                                         │   │
│   └─────────────────────────────────────────────────────────────┘   │
│                              │                                       │
│                              ▼                                       │
│   ┌─────────────────────────────────────────────────────────────┐   │
│   │                      快速恢复 (Fast Recovery)                 │   │
│   │                                                             │   │
│   │  1. ssthresh = CWND * 0.7                                  │   │
│   │  2. CWND = ssthresh + 3 * MSS (裂缝修复)                   │   │
│   │  3. 重传丢失的包                                            │   │
│   │  4. 收到 ACK → CWND += MSS (部分ACK增加)                    │   │
│   │  5. CWND 回到 ssthresh 后进入拥塞避免                      │   │
│   └─────────────────────────────────────────────────────────────┘   │
│                                                                      │
└──────────────────────────────────────────────────────────────────────┘
```

### 6.3 BBR 算法

```
BBR (Bottleneck Bandwidth and Round-trip propagation time):

┌──────────────────────────────────────────────────────────────────────┐
│                        BBR 原理                                     │
├──────────────────────────────────────────────────────────────────────┤
│                                                                      │
│   核心思想：                                                        │
│   · 估计可用带宽 (BtlBw)                                           │
│   · 估计最小 RTT (RTprop)                                          │
│   · 吞吐量 = BtlBw / RTprop                                        │
│                                                                      │
│   关键测量：                                                        │
│   · BtlBw: 通过吞吐量测量                                          │
│   · RTprop: 通过 RTT 测量                                          │
│                                                                      │
└──────────────────────────────────────────────────────────────────────┘

BBR 状态机：

┌──────────────────────────────────────────────────────────────────────┐
│                        BBR 状态机                                   │
├──────────────────────────────────────────────────────────────────────┤
│                                                                      │
│   ┌─────────┐                                                    │
│   │  STARTUP│  指数增长，发现瓶颈带宽                            │
│   └────┬────┘                                                    │
│        │ Draining                                               │
│        ▼                                                        │
│   ┌─────────┐  probe_bw                                          │
│   │  DRAIN │  排空 queues                                      │
│   └────┬────┘    │                                              │
│        │          ▼                                              │
│        │     ┌──────────────┐                                   │
│        │     │  PROBE_BW   │  周期性地增加/减少                  │
│        │     └──────┬───────┘  以探测更多带宽                   │
│        │            │                                           │
│        │      RTprop expired?                                   │
│        ▼            ▼                                            │
│   ┌─────────┐  ┌──────────────┐                                  │
│   │PROBE_RTT│  │  PROBE_BW   │                                  │
│   └─────────┘  └──────────────┘                                  │
│                                                                      │
│   RTprop 过期（> 10s 无新 RTprop）：                              │
│   · 进入 PROBE_RTT                                                │
│   · CWND = 4 * MSS                                               │
│   · RTprop 更新后恢复到 PROBE_BW                                  │
│                                                                      │
└──────────────────────────────────────────────────────────────────────┘
```

## 7. 丢包检测

### 7.1 丢包检测方法

```
QUIC 丢包检测方法：

┌──────────────────────────────────────────────────────────────────────┐
│                        丢包检测方法                                  │
├──────────────────────────────────────────────────────────────────────┤
│                                                                      │
│   1. PTO (Probe Timeout)                                           │
│      · 超时未确认，触发重传                                         │
│      · 基于计时器                                                   │
│                                                                      │
│   2. 包的 ACK 时间超过 PTO                                         │
│      · 等待 ACK 超时                                               │
│                                                                      │
│   3. 包的序号大于 已确认最大 + ack_eliciting 后                     │
│      · 丢包检测阈值                                                │
│                                                                      │
│   4. ECN (Explicit Congestion Notification)                         │
```

### 7.2 PTO 计算

```c
// PTO (Probe Timeout) 计算

struct pto_state {
    u64 duration;           // 当前 PTO 持续时间
    u64 probe_count;        // 探测次数
    bool handshakeconfirmed;// 握手是否确认
};

// 计算 PTO
u64 calculate_pto(struct pto_state *pto) {
    // PTO = max(kGranularity, SRTT + max(4*RTTVAR, kGranularity)) * 2
    // 初始值: 1 second

    u64 kGranularity = 1;  // ms
    u64 kDefaultPTO = 1000; // 1 second

    if (pto->handshake_confirmed) {
        // 握手确认后使用标准公式
        return max(kGranularity,
               pto->smoothed_rtt + max(4 * pto->rttvar, kGranularity)
           ) * 2;
    } else {
        // 握手期间使用更大值
        return kDefaultPTO * 3;
    }
}

// PTO 定时器
void pto_timer_check(struct connection *conn) {
    u64 now = get_time();
    u64 pto = calculate_pto(&conn->pto_state);

    if (now - conn->last_ack_sent > pto) {
        // PTO 过期，触发探测
        send_probe_packet(conn);
        conn->pto_state.probe_count++;
    }
}
```

### 7.3 丢包判定

```c
// 丢包判定逻辑

void on_ack_received(struct connection *conn, struct ack_frame *ack) {
    for each acked_packet in ack:
        // 移除 retransmit timer
        cancel_retransmit_timer(acked_packet);

        // 调用拥塞控制
        conn->cc.on_packets_acked(acked_packet);

    // 检查是否有包需要标记为丢失
    for each packet in conn->sent_packets:
        if (packet.pn < ack.largest_acked - REJECT_THRESHOLD) {
            // 超过阈值仍未确认，判定为丢失
            mark_packet_lost(packet);
            conn->cc.on_packets_lost(packet);
        }
}

// 丢包后处理
void mark_packet_lost(struct packet *pkt) {
    // 1. 从 sent 列表移除
    remove_from_sent_list(pkt);

    // 2. 触发重传
    if (pkt->is_ack_eliciting) {
        // 重传数据
        retransmit_data(pkt->stream_data);
    }

    // 3. 更新相关计时器
    update_retransmit_timer();
}
```

## 8. RTT 测量

### 8.1 RTT 测量原理

```
RTT 测量：

┌──────────────────────────────────────────────────────────────────────┐
│                        RTT 测量                                     │
├──────────────────────────────────────────────────────────────────────┤
│                                                                      │
│   发送 Packet N                                                     │
│   ┌────────────────────────────────────────────────────────────┐    │
│   │ Time: T1                                                   │    │
│   └────────────────────────────────────────────────────────────┘    │
│                              │                                       │
│                              │ (网络传输)                            │
│                              ▼                                       │
│                          Server                                      │
│                              │                                       │
│                              │ (处理)                                │
│                              ▼                                       │
│   ACK(N) ◄───────────────────────────────────────────────────────   │
│   ┌────────────────────────────────────────────────────────────┐    │
│   │ Time: T2                                                   │    │
│   └────────────────────────────────────────────────────────────┘    │
│                                                                      │
│   RTT = T2 - T1                                                    │
│                                                                      │
│   但 ACK 可能延迟，引入误差                                          │
│                                                                      │
└──────────────────────────────────────────────────────────────────────┘

RTT 计算：

  // 原始 RTT
  rtt = ack_time - send_time

  // 调整后的 RTT
  // 减去 ack_delay
  adjusted_rtt = rtt - ack_delay

  // SRTT (Smoothed RTT)
  // RFC 793 公式
  SRTT = (7/8) * SRTT + (1/8) * rtt

  // RTTVAR (RTT Variance)
  RTTVAR = (3/4) * RTTVAR + (1/4) * |SRTT - rtt|
```

### 8.2 ACK 延迟

```
ACK Delay：

┌──────────────────────────────────────────────────────────────────────┐
│                        ACK Delay                                    │
├──────────────────────────────────────────────────────────────────────┤
│                                                                      │
│   接收方延迟确认以提高效率：                                         │
│   · 不是每个包都立即 ACK                                            │
│   · 延迟一段时间，批量确认                                          │
│   · 减少 ACK 流量                                                   │
│                                                                      │
│   ACK Frame 包含 ack_delay 字段：                                   │
│   ┌────────────────────────────────────────────────────────────┐     │
│   │  Type: 0x02-0x03                                        │     │
│   │  Largest Acknowledged: varint                            │     │
│   │  ACK Delay: varint                                      │     │
│   │  ACK Ranges...                                          │     │
│   └────────────────────────────────────────────────────────────┘     │
│                                                                      │
│   接收方设置：                                                      │
│   ack_delay = min( max_ack_delay, current_time - packet_arrival ) │
│                                                                      │
│   发送方计算：                                                      │
│   adjusted_rtt = ack_time - send_time - decode_ack_delay()        │
│                                                                      │

max_ack_delay 限制：

  · 发送方通告 max_ack_delay
  · 接收方不能超过此值

  · 用于 PTO 计算
```

### 9.2 ACK 策略

```
ACK 策略：

┌──────────────────────────────────────────────────────────────────────┐
│                        ACK 策略                                     │
├──────────────────────────────────────────────────────────────────────┤
│                                                                      │
│   1. 每 N 个包 ACK 一次                                            │

TCP vs QUIC ACK 对比：

  TCP ACK:
  · 每个包一个 ACK（累计）
```
