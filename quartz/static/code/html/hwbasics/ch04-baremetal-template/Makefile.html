# ch04-baremetal-template -- Cortex-M3 裸机最小模板（QEMU mps2-an385）
#
#   make            编译 + 链接（产物在 build/）
#   make run        QEMU 运行（Ctrl-A X 退出）
#   make disasm     反汇编 + 向量表摘录
#   make gdb        QEMU 起 gdbserver 等待远程调试（另终端 arm 侧 gdb 连 :1234）
#   make clean      清理

CROSS   ?= arm-none-eabi-
CC       := $(CROSS)gcc
OBJDUMP  := $(CROSS)objdump
SIZE     := $(CROSS)size

TARGET   := ch04-baremetal
BUILD    := build
ELF      := $(BUILD)/$(TARGET).elf

# Cortex-M3 只有 Thumb-2，无 ARM 状态；-mthumb 是硬性要求
MCUFLAGS := -mcpu=cortex-m3 -mthumb
CFLAGS   := $(MCUFLAGS) -Og -g3 -ffreestanding -fno-common \
            -Wall -Wextra -ffunction-sections -fdata-sections
# -nostdlib：不要 crt0/标准库——启动文件我们自己写
# -T：指定链接脚本；-Wl,-Map：输出 map 文件
LDFLAGS  := $(MCUFLAGS) -nostdlib -T mps2_an385.ld \
            -Wl,--gc-sections -Wl,-Map=$(BUILD)/$(TARGET).map

QEMU     ?= qemu-system-arm
QEMUFLAGS := -M mps2-an385 -kernel $(ELF) -nographic

SRCS_S   := startup.S
SRCS_C   := main.c
OBJS     := $(patsubst %.S,$(BUILD)/%.o,$(SRCS_S)) \
            $(patsubst %.c,$(BUILD)/%.o,$(SRCS_C))

.PHONY: all run disasm gdb clean

all: $(ELF)
	$(SIZE) $(ELF)

$(BUILD):
	mkdir -p $(BUILD)

$(BUILD)/%.o: %.S | $(BUILD)
	$(CC) $(CFLAGS) -c -o $@ $<

$(BUILD)/%.o: %.c | $(BUILD)
	$(CC) $(CFLAGS) -c -o $@ $<

$(ELF): $(OBJS) mps2_an385.ld
	$(CC) $(LDFLAGS) -o $@ $(OBJS)

run: $(ELF)
	$(QEMU) $(QEMUFLAGS)

disasm: $(ELF)
	$(OBJDUMP) -d $(ELF) | head -40
	@echo '... (完整反汇编见 make disasm | less)'

gdb: $(ELF)
	$(QEMU) $(QEMUFLAGS) -S -s

clean:
	rm -rf $(BUILD)
