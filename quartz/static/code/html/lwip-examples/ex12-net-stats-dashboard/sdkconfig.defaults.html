CONFIG_ETH_USE_OPENETH=y

# 五段协议计数增量打印（EX12-ST 行）依赖它；关闭后示例仍可运行，只是统计段恒空。

CONFIG_LWIP_STATS=y

# IDF v6 默认关；xCoreID 字段要 VTASKLIST_INCLUDE_COREID 才编入 TaskStatus_t
# （依赖 STATS_FORMATTING_FUNCTIONS）。不开则任务表段降级为 disabled 提示。

CONFIG_FREERTOS_USE_TRACE_FACILITY=y
CONFIG_FREERTOS_USE_STATS_FORMATTING_FUNCTIONS=y
CONFIG_FREERTOS_VTASKLIST_INCLUDE_COREID=y
