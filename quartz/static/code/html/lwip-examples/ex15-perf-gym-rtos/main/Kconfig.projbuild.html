        int "Fault scenario (0=healthy baseline 1=prio-starve 2=slow-raw-callback 3=mbox-flood 4=stack-hwm)"

        default 0
        help
            0 健康基线；1 双核高优自旋饿死低优心跳（ch19 占空比实验）；

            3 高频投递打满 32 槽 tcpip 邮箱（ch13 实验 D 应用版）；
            4 任务栈余量耗尽告警教学（HWM 观测，不含真溢出崩溃）。

    config EX15_SC1_DUTY_PCT
        int "SC1 spinner duty cycle (%) applied on both pinned cores"
        range 50 100
        default 100
        help
            ch19 实测分水岭：单核钉核自旋 duty<=70% 存活、70%~85% 引爆、100% 全灭。

            想亲手踩 70%~85% 分水岭请调低后重测。

    config EX15_SC3_WORKERS

        default 3

    config EX15_SC3_BURST_N
        int "SC3 capacity-probe burst size (expect first N-32 accepted)"
        range 33 256

    config EX15_SC3_WORK_US
        int "SC3 per-callback work in tcpip thread (us)"
        range 0 4000
        default 400
        help
            让被淹没端"消化变慢"的诚实模拟：真实系统投进邮箱的消息本就带工作量。
            0 = 空回调（消化 ~22µs/条，ch13 口径），难以稳定打满。

    config EX15_SC3_PACE_MS
        int "SC3 forced sleep between posts per worker (ms)"
        range 1 20
        default 1
        help
            生产者礼貌性节流：不带它，永不阻塞的 flood 循环会把低于自己优先级
            的心跳任务一并饿死，污染本场景的"纯邮箱"叙事（那属于场景 1）。

    config EX15_SC4_STACK_BYTES
        int "SC4 victim task stack size (bytes)"
        range 1024 8192
        default 4096

    config EX15_SC4_CHUNK
        int "SC4 per-level local array chunk (bytes)"
        range 64 512
        default 256

    config EX15_SC4_ALARM_HWM
        int "SC4 high-water alarm threshold: free bytes below this fires kind=stk"
        range 384 2048
        default 1024
        help
            告警要早于危险：阈值必须高于绝对安全地板（EX15_SC4_STOP_MARGIN，
            默认 768），否则永远轮不到告警就已经踩到护栏。

    config EX15_SC4_STOP_MARGIN
        int "SC4 absolute safety floor: stop descending at this much free"
        range 384 3072
        default 768
        help
            实测教训：首版地板设 192B 时，步进之间的 printf（自身要用几百字节栈）
            直接把栈底打穿——backtrace 里出现 0xa5a5a5a5 CORRUPTED。地板必须为
            "你要在栈上继续干的活"留足余量。
