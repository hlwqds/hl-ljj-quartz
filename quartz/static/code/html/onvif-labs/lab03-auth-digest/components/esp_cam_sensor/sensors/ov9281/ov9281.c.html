typedef struct {
    uint32_t exposure_val;
    uint32_t exposure_max;
    uint32_t gain_index; // current gain index

    uint32_t vflip_en : 1;
    uint32_t hmirror_en : 1;
} ov9281_para_t;

struct ov9281_cam {
    ov9281_para_t ov9281_para;
};

static esp_err_t ov9281_read(esp_sccb_io_handle_t sccb_handle, uint16_t reg, uint8_t *read_buf)
{
    return esp_sccb_transmit_receive_reg_a16v8(sccb_handle, reg, read_buf);
}

static esp_err_t ov9281_write(esp_sccb_io_handle_t sccb_handle, uint16_t reg, uint8_t data)
{
    return esp_sccb_transmit_reg_a16v8(sccb_handle, reg, data);
}

/* write a array of registers  */
static esp_err_t ov9281_write_array(esp_sccb_io_handle_t sccb_handle, ov9281_reginfo_t *regarray)
{
    int i = 0;
    esp_err_t ret = ESP_OK;
    while ((ret == ESP_OK) && regarray[i].reg != OV9281_REG_END) {
        if (regarray[i].reg != OV9281_REG_DELAY) {
            ret = ov9281_write(sccb_handle, regarray[i].reg, regarray[i].val);
        } else {
            delay_ms(regarray[i].val);
        }
        i++;
    }
    ESP_LOGD(TAG, "Set array done[i=%d]", i);
    return ret;
}

static esp_err_t ov9281_set_reg_bits(esp_sccb_io_handle_t sccb_handle, uint16_t reg, uint8_t offset, uint8_t length, uint8_t value)
{
    esp_err_t ret = ESP_OK;
    uint8_t reg_data = 0;

    ret = ov9281_read(sccb_handle, reg, &reg_data);
    if (ret != ESP_OK) {
        return ret;
    }
    uint8_t mask = ((1 << length) - 1) << offset;
    value = (reg_data & ~mask) | ((value << offset) & mask);
    ret = ov9281_write(sccb_handle, reg, value);
    return ret;
}

static esp_err_t ov9281_set_test_pattern(esp_cam_sensor_device_t *dev, int enable)
{
    return ov9281_set_reg_bits(dev->sccb_handle, OV9281_REG_TEST_PATTERN, 7, 1,  enable ? 0x01 : 0x00);
}

static esp_err_t ov9281_hw_reset(esp_cam_sensor_device_t *dev)
{
    if (dev->reset_pin >= 0) {
        gpio_set_level(dev->reset_pin, 0);
        delay_ms(10);
        gpio_set_level(dev->reset_pin, 1);
        delay_ms(10);
    }
    return ESP_OK;
}

static esp_err_t ov9281_soft_reset(esp_cam_sensor_device_t *dev)
{
    esp_err_t ret = ov9281_write(dev->sccb_handle, OV9281_REG_SW_RESET, 0x01);
    delay_ms(5);
    return ret;
}

static esp_err_t ov9281_get_sensor_id(esp_cam_sensor_device_t *dev, esp_cam_sensor_id_t *id)
{
    esp_err_t ret = ESP_FAIL;
    uint8_t pid_h, pid_l;

    ret = ov9281_read(dev->sccb_handle, OV9281_REG_CHIP_ID_H, &pid_h);
    if (ret != ESP_OK) {
        return ret;
    }
    ret = ov9281_read(dev->sccb_handle, OV9281_REG_CHIP_ID_L, &pid_l);
    if (ret != ESP_OK) {
        return ret;
    }
    id->pid = (pid_h << 8) | pid_l;

    return ret;
}

static esp_err_t ov9281_set_stream(esp_cam_sensor_device_t *dev, int enable)
{
    esp_err_t ret = ESP_FAIL;
    ret = ov9281_write(dev->sccb_handle, OV9281_REG_CTRL_MODE, enable ? OV9281_MODE_STREAMING : OV9281_MODE_SW_STANDBY);

    dev->stream_status = enable;
    ESP_LOGD(TAG, "Stream=%d", enable);
    return ret;
}

static esp_err_t ov9281_set_mirror(esp_cam_sensor_device_t *dev, int enable)
{
    return ov9281_set_reg_bits(dev->sccb_handle, 0x3820, 2, 1,  enable ? 0x01 : 0x00);
}

static esp_err_t ov9281_set_vflip(esp_cam_sensor_device_t *dev, int enable)
{
    return ov9281_set_reg_bits(dev->sccb_handle, 0x3821, 2, 1,  enable ? 0x01 : 0x00);
}

static esp_err_t ov9281_set_exp_val(esp_cam_sensor_device_t *dev, uint32_t u32_val)
{
    esp_err_t ret;
    struct ov9281_cam *cam_ov9281 = (struct ov9281_cam *)dev->priv;
    uint32_t value_buf = MAX(u32_val, s_ov9281_exp_min);
    value_buf = MIN(value_buf, cam_ov9281->ov9281_para.exposure_max);

    ESP_LOGD(TAG, "set exposure 0x%" PRIx32, value_buf);
    /* 4 least significant bits of expsoure are fractional part */
    ret = ov9281_write(dev->sccb_handle,
                       OV9281_REG_EXPOSURE_H,
                       OV9281_FETCH_EXP_H(value_buf));
    ret |= ov9281_write(dev->sccb_handle,
                        OV9281_REG_EXPOSURE_M,
                        OV9281_FETCH_EXP_M(value_buf));
    ret |= ov9281_write(dev->sccb_handle,
                        OV9281_REG_EXPOSURE_L,
                        OV9281_FETCH_EXP_L(value_buf));
    if (ret == ESP_OK) {
        cam_ov9281->ov9281_para.exposure_val = value_buf;
    }
    return ret;
}

static esp_err_t ov9281_set_total_gain_val(esp_cam_sensor_device_t *dev, uint32_t u32_val)
{
    esp_err_t ret;
    struct ov9281_cam *cam_ov9281 = (struct ov9281_cam *)dev->priv;
    u32_val = MIN(u32_val, s_ov9281_limited_gain_index);

    ESP_LOGD(TAG, "dgain %" PRIx8, ov9281_gain_reg_map[u32_val]);
    ret = ov9281_write(dev->sccb_handle,
                       OV9281_REG_GAIN,
                       ov9281_gain_reg_map[u32_val]);
    if (ret == ESP_OK) {
        cam_ov9281->ov9281_para.gain_index = u32_val;
    }
    return ret;
}

static esp_err_t ov9281_query_para_desc(esp_cam_sensor_device_t *dev, esp_cam_sensor_param_desc_t *qdesc)
{
    esp_err_t ret = ESP_OK;
    switch (qdesc->id) {
    case ESP_CAM_SENSOR_EXPOSURE_VAL:
        qdesc->type = ESP_CAM_SENSOR_PARAM_TYPE_NUMBER;
        qdesc->number.minimum = s_ov9281_exp_min;
        qdesc->number.maximum = dev->cur_format->isp_info->isp_v1_info.vts - OV9281_EXP_MAX_OFFSET; // max = VTS-6 = height+vblank-6, so when update vblank, exposure_max must be updated
        qdesc->number.step = 1;
        qdesc->default_value = dev->cur_format->isp_info->isp_v1_info.exp_def;
        break;
    case ESP_CAM_SENSOR_EXPOSURE_US:
        qdesc->type = ESP_CAM_SENSOR_PARAM_TYPE_NUMBER;
        qdesc->number.minimum = EXPOSURE_OV9281_TO_V4L2(s_ov9281_exp_min, dev->cur_format);
        qdesc->number.maximum = EXPOSURE_OV9281_TO_V4L2((dev->cur_format->isp_info->isp_v1_info.vts - OV9281_EXP_MAX_OFFSET), dev->cur_format); // max = VTS-6 = height+vblank-6, so when update vblank, exposure_max must be updated
        qdesc->number.step = MAX(EXPOSURE_OV9281_TO_V4L2(0x01, dev->cur_format), 1);
        qdesc->default_value = EXPOSURE_OV9281_TO_V4L2((dev->cur_format->isp_info->isp_v1_info.exp_def), dev->cur_format);
        break;
    case ESP_CAM_SENSOR_GAIN:
        qdesc->type = ESP_CAM_SENSOR_PARAM_TYPE_ENUMERATION;
        qdesc->enumeration.count = s_ov9281_limited_gain_index;
        qdesc->enumeration.elements = ov9281_total_gain_val_map;
        qdesc->default_value = dev->cur_format->isp_info->isp_v1_info.gain_def; // gain index
        break;
    case ESP_CAM_SENSOR_GROUP_EXP_GAIN:
        qdesc->type = ESP_CAM_SENSOR_PARAM_TYPE_U8;
        qdesc->u8.size = sizeof(esp_cam_sensor_gh_exp_gain_t);
        break;
    case ESP_CAM_SENSOR_VFLIP:
    case ESP_CAM_SENSOR_HMIRROR:
        qdesc->type = ESP_CAM_SENSOR_PARAM_TYPE_NUMBER;
        qdesc->number.minimum = 0;
        qdesc->number.maximum = 1;
        qdesc->number.step = 1;
        qdesc->default_value = 0;
        break;
    default: {
        ESP_LOGD(TAG, "id=%"PRIx32" is not supported", qdesc->id);
        ret = ESP_ERR_INVALID_ARG;
        break;
    }
    }
    return ret;
}

static esp_err_t ov9281_get_para_value(esp_cam_sensor_device_t *dev, uint32_t id, void *arg, size_t size)
{
    esp_err_t ret = ESP_OK;
    struct ov9281_cam *cam_ov9281 = (struct ov9281_cam *)dev->priv;
    switch (id) {
    case ESP_CAM_SENSOR_EXPOSURE_VAL: {
        *(uint32_t *)arg = cam_ov9281->ov9281_para.exposure_val;
        break;
    }
    case ESP_CAM_SENSOR_GAIN: {
        *(uint32_t *)arg = cam_ov9281->ov9281_para.gain_index;
        break;
    }
    default: {
        ret = ESP_ERR_NOT_SUPPORTED;
        break;
    }
    }
    return ret;
}

static esp_err_t ov9281_set_para_value(esp_cam_sensor_device_t *dev, uint32_t id, const void *arg, size_t size)
{
    esp_err_t ret = ESP_OK;

    switch (id) {
    case ESP_CAM_SENSOR_EXPOSURE_VAL: {
        uint32_t u32_val = *(uint32_t *)arg;
        // Todo, Add group hold if fast AEC need
        ret = ov9281_set_exp_val(dev, u32_val);
        break;
    }
    case ESP_CAM_SENSOR_EXPOSURE_US: {
        uint32_t u32_val = *(uint32_t *)arg;
        uint32_t ori_exp = EXPOSURE_V4L2_TO_OV9281(u32_val, dev->cur_format);
        ret = ov9281_set_exp_val(dev, ori_exp);
        break;
    }
    case ESP_CAM_SENSOR_GAIN: {
        uint32_t u32_val = *(uint32_t *)arg;
        ret = ov9281_set_total_gain_val(dev, u32_val);
        break;
    }
    case ESP_CAM_SENSOR_GROUP_EXP_GAIN: {
        esp_cam_sensor_gh_exp_gain_t *value = (esp_cam_sensor_gh_exp_gain_t *)arg;
        uint32_t ori_exp = 0;
        if (value->exposure_us != 0) {
            ori_exp = EXPOSURE_V4L2_TO_OV9281(value->exposure_us, dev->cur_format);
        } else if (value->exposure_val != 0) {
            ori_exp = value->exposure_val;
        } else {
            ret = ESP_ERR_INVALID_ARG;
            break;
        }
        ret = ov9281_set_exp_val(dev, ori_exp);
        ret |= ov9281_set_total_gain_val(dev, value->gain_index);
        break;
    }
    case ESP_CAM_SENSOR_VFLIP: {
        int *value = (int *)arg;
        ret = ov9281_set_vflip(dev, *value);
        break;
    }
    case ESP_CAM_SENSOR_HMIRROR: {
        int *value = (int *)arg;
        ret = ov9281_set_mirror(dev, *value);
        break;
    }
    default: {
        ESP_LOGE(TAG, "set id=%" PRIx32 " is not supported", id);
        ret = ESP_ERR_INVALID_ARG;
        break;
    }
    }

    return ret;
}

static esp_err_t ov9281_query_support_formats(esp_cam_sensor_device_t *dev, esp_cam_sensor_format_array_t *formats)
{
    ESP_CAM_SENSOR_NULL_POINTER_CHECK(TAG, dev);
    ESP_CAM_SENSOR_NULL_POINTER_CHECK(TAG, formats);

    formats->count = ARRAY_SIZE(ov9281_format_info);
    formats->format_array = &ov9281_format_info[0];
    return ESP_OK;
}

static esp_err_t ov9281_query_support_capability(esp_cam_sensor_device_t *dev, esp_cam_sensor_capability_t *sensor_cap)
{
    ESP_CAM_SENSOR_NULL_POINTER_CHECK(TAG, dev);
    ESP_CAM_SENSOR_NULL_POINTER_CHECK(TAG, sensor_cap);

    sensor_cap->fmt_raw = 1;
    return 0;
}

static esp_err_t ov9281_set_format(esp_cam_sensor_device_t *dev, const esp_cam_sensor_format_t *format)
{
    ESP_CAM_SENSOR_NULL_POINTER_CHECK(TAG, dev);
    struct ov9281_cam *cam_ov9281 = (struct ov9281_cam *)dev->priv;
    esp_err_t ret = ESP_OK;
    /* Depending on the interface type, an available configuration is automatically loaded.
    You can set the output format of the sensor without using query_format().*/
    if (format == NULL) {
        format = &ov9281_format_info[CONFIG_CAMERA_OV9281_MIPI_IF_FORMAT_INDEX_DEFAULT];
    }

    ret = ov9281_write_array(dev->sccb_handle, (ov9281_reginfo_t *)format->regs);
    ESP_LOGD(TAG, "%s", format->name);

    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "Set format regs fail");
        return ESP_CAM_SENSOR_ERR_FAILED_SET_FORMAT;
    }

    dev->cur_format = format;
    // init para
    cam_ov9281->ov9281_para.exposure_val = dev->cur_format->isp_info->isp_v1_info.exp_def;
    cam_ov9281->ov9281_para.gain_index = dev->cur_format->isp_info->isp_v1_info.gain_def;
    cam_ov9281->ov9281_para.exposure_max = dev->cur_format->isp_info->isp_v1_info.vts - OV9281_EXP_MAX_OFFSET;

    return ret;
}

static esp_err_t ov9281_get_format(esp_cam_sensor_device_t *dev, esp_cam_sensor_format_t *format)
{
    ESP_CAM_SENSOR_NULL_POINTER_CHECK(TAG, dev);
    ESP_CAM_SENSOR_NULL_POINTER_CHECK(TAG, format);

    esp_err_t ret = ESP_FAIL;

    if (dev->cur_format != NULL) {
        memcpy(format, dev->cur_format, sizeof(esp_cam_sensor_format_t));
        ret = ESP_OK;
    }
    return ret;
}

static esp_err_t ov9281_priv_ioctl(esp_cam_sensor_device_t *dev, uint32_t cmd, void *arg)
{
    esp_err_t ret = ESP_OK;
    uint8_t regval;
    esp_cam_sensor_reg_val_t *sensor_reg;
    OV9281_IO_MUX_LOCK(mux);

    switch (cmd) {
    case ESP_CAM_SENSOR_IOC_HW_RESET:
        ret = ov9281_hw_reset(dev);
        break;
    case ESP_CAM_SENSOR_IOC_SW_RESET:
        ret = ov9281_soft_reset(dev);
        break;
    case ESP_CAM_SENSOR_IOC_S_REG:
        sensor_reg = (esp_cam_sensor_reg_val_t *)arg;
        ret = ov9281_write(dev->sccb_handle, sensor_reg->regaddr, sensor_reg->value);
        break;
    case ESP_CAM_SENSOR_IOC_S_STREAM:
        ret = ov9281_set_stream(dev, *(int *)arg);
        break;
    case ESP_CAM_SENSOR_IOC_S_TEST_PATTERN:
        ret = ov9281_set_test_pattern(dev, *(int *)arg);
        break;
    case ESP_CAM_SENSOR_IOC_G_REG:
        sensor_reg = (esp_cam_sensor_reg_val_t *)arg;
        ret = ov9281_read(dev->sccb_handle, sensor_reg->regaddr, &regval);
        if (ret == ESP_OK) {
            sensor_reg->value = regval;
        }
        break;
    case ESP_CAM_SENSOR_IOC_G_CHIP_ID:
        ret = ov9281_get_sensor_id(dev, arg);
        break;
    default:
        break;
    }

    OV9281_IO_MUX_UNLOCK(mux);
    return ret;
}

static esp_err_t ov9281_power_on(esp_cam_sensor_device_t *dev)
{
    esp_err_t ret = ESP_OK;

    if (dev->xclk_pin >= 0) {
        OV9281_ENABLE_OUT_XCLK(dev->xclk_pin, dev->xclk_freq_hz);
    }

    if (dev->pwdn_pin >= 0) {
        gpio_config_t conf = { 0 };
        conf.pin_bit_mask = 1LL << dev->pwdn_pin;
        conf.mode = GPIO_MODE_OUTPUT;
        gpio_config(&conf);

        // carefully, logic is inverted compared to reset pin
        gpio_set_level(dev->pwdn_pin, 1);
        delay_ms(10);
        gpio_set_level(dev->pwdn_pin, 0);
        delay_ms(10);
    }

    if (dev->reset_pin >= 0) {
        gpio_config_t conf = { 0 };
        conf.pin_bit_mask = 1LL << dev->reset_pin;
        conf.mode = GPIO_MODE_OUTPUT;
        gpio_config(&conf);

        gpio_set_level(dev->reset_pin, 0);
        delay_ms(10);
        gpio_set_level(dev->reset_pin, 1);
        delay_ms(10);
    }

    return ret;
}

static esp_err_t ov9281_power_off(esp_cam_sensor_device_t *dev)
{
    esp_err_t ret = ESP_OK;

    if (dev->xclk_pin >= 0) {
        OV9281_DISABLE_OUT_XCLK(dev->xclk_pin);
    }

    if (dev->pwdn_pin >= 0) {
        gpio_set_level(dev->pwdn_pin, 0);
        delay_ms(10);
        gpio_set_level(dev->pwdn_pin, 1);
        delay_ms(10);
    }

    if (dev->reset_pin >= 0) {
        gpio_set_level(dev->reset_pin, 1);
        delay_ms(10);
        gpio_set_level(dev->reset_pin, 0);
        delay_ms(10);
    }

    return ret;
}

static esp_err_t ov9281_delete(esp_cam_sensor_device_t *dev)
{
    ESP_LOGD(TAG, "del ov9281 (%p)", dev);
    if (dev) {
        if (dev->priv) {
            free(dev->priv);
            dev->priv = NULL;
        }
        free(dev);
        dev = NULL;
    }

    return ESP_OK;
}

static const esp_cam_sensor_ops_t ov9281_ops = {
    .query_para_desc = ov9281_query_para_desc,
    .get_para_value = ov9281_get_para_value,
    .set_para_value = ov9281_set_para_value,
    .query_support_formats = ov9281_query_support_formats,
    .query_support_capability = ov9281_query_support_capability,
    .set_format = ov9281_set_format,
    .get_format = ov9281_get_format,
    .priv_ioctl = ov9281_priv_ioctl,
    .del = ov9281_delete
};

#if OV9281_EXPOSURE_TEST_EN
static uint32_t s_exp_v = 0x05;
static uint32_t s_exp_add = true;
TimerHandle_t ae_timer_handle;
static void ae_timer_callback(TimerHandle_t timer)
{
    esp_cam_sensor_device_t *dev = (esp_cam_sensor_device_t *)pvTimerGetTimerID(timer);
    struct ov9281_cam *cam_ov9281 = (struct ov9281_cam *)dev->priv;
#if OV9281_EXPOSURE_TEST_EN_GAIN
    if (s_exp_v >= s_ov9281_limited_gain_index) {
        s_exp_add = false;
    } else if (s_exp_v < 1) {
        s_exp_add = true;
    }
    ov9281_set_total_gain_val(dev, s_exp_v);
    if (s_exp_add == true) {
        s_exp_v++;
    } else {
        s_exp_v--;
    }
#else
    if (s_exp_v >= cam_ov9281->ov9281_para.exposure_max) {
        s_exp_add = false;
    } else if (s_exp_v < s_ov9281_exp_min) {
        s_exp_add = true;
    }
    ov9281_set_exp_val(dev, s_exp_v);
    if (s_exp_add == true) {
        s_exp_v++;
    } else {
        s_exp_v--;
    }
#endif
    ESP_LOGI(TAG, "E=%" PRIu32, s_exp_v);
}
#endif

esp_cam_sensor_device_t *ov9281_detect(esp_cam_sensor_config_t *config)
{
    esp_cam_sensor_device_t *dev = NULL;
    struct ov9281_cam *cam_ov9281;
    s_ov9281_limited_gain_index = ARRAY_SIZE(ov9281_total_gain_val_map);
    if (config == NULL) {
        return NULL;
    }

    dev = calloc(1, sizeof(esp_cam_sensor_device_t));
    if (dev == NULL) {
        ESP_LOGE(TAG, "No memory for camera");
        return NULL;
    }

    cam_ov9281 = heap_caps_calloc(1, sizeof(struct ov9281_cam), MALLOC_CAP_DEFAULT);
    if (!cam_ov9281) {
        ESP_LOGE(TAG, "failed to calloc cam");
        free(dev);
        return NULL;
    }

    dev->name = (char *)OV9281_SENSOR_NAME;
    dev->sccb_handle = config->sccb_handle;
    dev->xclk_pin = config->xclk_pin;
    dev->reset_pin = config->reset_pin;
    dev->pwdn_pin = config->pwdn_pin;
    dev->sensor_port = config->sensor_port;
    dev->ops = &ov9281_ops;
    dev->cur_format = &ov9281_format_info[CONFIG_CAMERA_OV9281_MIPI_IF_FORMAT_INDEX_DEFAULT];
    // init para
    cam_ov9281->ov9281_para.exposure_val = dev->cur_format->isp_info->isp_v1_info.exp_def;
    cam_ov9281->ov9281_para.gain_index = dev->cur_format->isp_info->isp_v1_info.gain_def;
    cam_ov9281->ov9281_para.exposure_max = dev->cur_format->isp_info->isp_v1_info.vts - OV9281_EXP_MAX_OFFSET;
    dev->priv = cam_ov9281;
    for (size_t i = 0; i < ARRAY_SIZE(ov9281_total_gain_val_map); i++) {
        if (ov9281_total_gain_val_map[i] > s_ov9281_limited_gain_) {
            s_ov9281_limited_gain_index = i - 1;
            break;
        }
    }

    // Configure sensor power, clock, and SCCB port
    if (ov9281_power_on(dev) != ESP_OK) {
        ESP_LOGE(TAG, "Camera power on failed");
        goto err_free_handler;
    }

    if (ov9281_get_sensor_id(dev, &dev->id) != ESP_OK) {
        ESP_LOGE(TAG, "Get sensor ID failed");
        goto err_free_handler;
    } else if (dev->id.pid != OV9281_PID) {
        ESP_LOGE(TAG, "Camera sensor is not OV9281, PID=0x%x", dev->id.pid);
        goto err_free_handler;
    }
    ESP_LOGI(TAG, "Detected Camera sensor PID=0x%x", dev->id.pid);

#if OV9281_EXPOSURE_TEST_EN
    ae_timer_handle = xTimerCreate("AE_t", 100 / portTICK_PERIOD_MS, pdTRUE,
                                   (void *)dev, ae_timer_callback);
    xTimerStart(ae_timer_handle, portMAX_DELAY);
#endif

    return dev;

err_free_handler:
    ov9281_power_off(dev);
    free(dev->priv);
    free(dev);

    return NULL;
}

{
    ((esp_cam_sensor_config_t *)config)->sensor_port = ESP_CAM_SENSOR_MIPI_CSI;
    return ov9281_detect(config);
