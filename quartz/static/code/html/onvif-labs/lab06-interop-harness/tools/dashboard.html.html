    <script>
      const $ = (id) => document.getElementById(id)
      let frameCount = 0,
        lastFrameTime = 0,
        fpsTimer = null

      // 连接状态
      const video = $("video")
      video.onload = () => {
        $("conn-badge").textContent = "STREAMING"
        $("conn-badge").className = "badge ok"
        frameCount++
        const now = Date.now()
        if (lastFrameTime && now - lastFrameTime > 0) {
          const fps = Math.round((frameCount / ((now - startTime) / 1000)) * 10) / 10
          $("fps").textContent = fps
        }
      }
      video.onerror = () => {
        $("conn-badge").textContent = "DISCONNECTED"
        $("conn-badge").className = "badge warn"
      }
      const startTime = Date.now()

      // Device Info
      fetch("/api/info")
        .then((r) => r.json())
        .then((d) => {
          $("dev-info").innerHTML = Object.entries(d)
            .map(([k, v]) => `<tr><td>${k}</td><td>${v}</td></tr>`)
            .join("")
        })
        .catch(() => {
          $("dev-info").innerHTML = "<tr><td>unavailable</td></tr>"
        })

      // Profile
      fetch("/api/profiles")
        .then((r) => r.json())
        .then((list) => {
          const p = list[0]
          if (!p) return
          $("profile-info").innerHTML = `
    <tr><td>Token</td><td>${p.token}</td></tr>
    <tr><td>Name</td><td>${p.name}</td></tr>
    <tr><td>Encoding</td><td>${p.encoding}</td></tr>
    <tr><td>Resolution</td><td>${p.width}×${p.height}</td></tr>`
        })
        .catch(() => {})

      // Stream URI
      fetch("/api/stream_uri")
        .then((r) => r.json())
        .then((d) => {
          $("stream-uri").textContent = d.Uri || "—"
        })
        .catch(() => {
          $("stream-uri").textContent = "unavailable"
        })

      // System Time
      fetch("/api/datetime")
        .then((r) => r.json())
        .then((d) => {
          $("dt-type").textContent = d.DateTimeType || "—"
          $("dt-utc").textContent = d.UTC || "—"
        })
        .catch(() => {})

      // Snapshot refresh (局部刷新，不影响活流)
      function refreshSnap() {
        const img = $("snap-img")
        const ts = Date.now()
        img.src = "/api/snapshot?ts=" + ts
        img.onload = () => {
          $("snap-status").textContent =
            `✓ ${new Date().toLocaleTimeString()} · ${img.naturalWidth}×${img.naturalHeight}`
        }
        img.onerror = () => {
          $("snap-status").textContent = "✗ fetch failed"
        }
      }

      // Board Metrics
      function refreshMetrics() {
        fetch("/api/status")
          .then((r) => r.json())
          .then((d) => {
            $("metrics").innerHTML = Object.entries(d)
              .map(
                ([k, v]) =>
                  `<div class="metric"><span class="label">${k}</span><span class="value">${v}</span></div>`,
              )
              .join("")
          })
          .catch(() => {})
      }
      refreshMetrics()
      setInterval(refreshMetrics, 5000)
    </script>
