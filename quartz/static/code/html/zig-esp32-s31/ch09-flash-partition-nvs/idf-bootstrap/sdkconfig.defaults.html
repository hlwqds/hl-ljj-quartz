# ch09-flash-partition-nvs / idf-bootstrap 的工程默认配置。
#
# 这个工程只产 bootloader 与分区表，但这两样必须和模组、应用一致：
#   1. 模组 ESP32-S31-WROOM-3（N16R16V）是 16 MB Flash；
#   2. 本章改用**自定义分区表** partitions.csv：factory 应用槽仍在 0x10000，
#      末尾多一个 storage 数据分区（0x300000 / 0x10000），给 KV 用。
#      tools/flash.sh 仍然按 flash_args 里的 0x10000 行替换应用镜像。
#
# 目标芯片不写在这里：idf.py --preview set-target esp32s31 会写进生成的
# sdkconfig（生成物，不入库）。sdkconfig.defaults 只在生成 sdkconfig 时生效，
# 已有 sdkconfig 时要先删除或走 menuconfig 重新确认。
#
# 本章不启用 PSRAM：分区表与 KV 都在内部 SRAM 的副本上解析、在 flash 上持久化，
# 与 PSRAM 无关。
#
# 控制台保持默认：应用侧 TX 走 ROM 的 esp_rom_printf，恒定输出到 UART0
# （GPIO58/59）、115200 8N1。

CONFIG_ESPTOOLPY_FLASHSIZE_16MB=y
CONFIG_ESPTOOLPY_FLASHSIZE="16MB"

# 自定义分区表：CONFIG_PARTITION_TABLE_CUSTOM 让构建读 CSV；
# FILENAME 是 cmake 侧实际使用的路径（相对工程根），两个一起写才稳。
CONFIG_PARTITION_TABLE_CUSTOM=y
CONFIG_PARTITION_TABLE_CUSTOM_FILENAME="partitions.csv"
CONFIG_PARTITION_TABLE_FILENAME="partitions.csv"
